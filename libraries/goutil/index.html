type Student struct {
  Name   string `json:"name"`
  RoleNo int    `json:"roleNo"`
}

student := Student{
  Name:   "Lucian",
  RoleNo: 7,
}

fmt.Println(goutil.ToJSON(student))
// out: {"name":"Lucian","roleNo":7}

fmt.Println(goutil.ToJSONIndent(student))
// out:
{
  "name": "Lucian",
  "roleNo": 7
}

goutil.PrintToJSON(student)
// out:
{
  "name": "Lucian",
  "roleNo": 7
}
